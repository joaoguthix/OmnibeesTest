-- Banco de Dados
CREATE DATABASE OmniBees;
GO
USE OmniBees;

CREATE TABLE Cotacao (
    Id INT NOT NULL PRIMARY KEY IDENTITY(1,1),
    IdProduto INT NOT NULL, -- Alterado para INT
    DataCriacao DATETIME NOT NULL,
    DataAtualizacao DATETIME NOT NULL,
    IdParceiro INT NOT NULL, -- Alterado para INT
    NomeSegurado NVARCHAR(100) NOT NULL,
    DDD NVARCHAR(5) NULL,
    Telefone NVARCHAR(15) NULL,
    Endereco NVARCHAR(255) NOT NULL,
    CEP NVARCHAR(10) NOT NULL,
    Documento NVARCHAR(20) NOT NULL,
    Nascimento DATETIME NOT NULL,
    Premio DECIMAL(18,2) NOT NULL,
    ImportanciaSegurada DECIMAL(18,2) NOT NULL
);
GO

CREATE TABLE CotacaoBeneficiario (
    Id INT NOT NULL PRIMARY KEY IDENTITY(1,1),
    IdCotacao INT NOT NULL, -- Alterado para INT
    Nome NVARCHAR(100) NOT NULL,
    Percentual DECIMAL(5,2) NOT NULL,
    IdParentesco INT NOT NULL, -- Alterado para INT

    CONSTRAINT FK_CotacaoBeneficiario_Cotacao FOREIGN KEY (IdCotacao)
        REFERENCES Cotacao(Id)
        ON DELETE CASCADE
);
GO

CREATE TABLE CotacaoCobertura (
    Id INT NOT NULL PRIMARY KEY IDENTITY(1,1),
    IdCotacao INT NOT NULL, -- Alterado para INT
    IdCobertura INT NOT NULL, -- Alterado para INT
    ValorDesconto DECIMAL(18,2) NULL,
    ValorAgravo DECIMAL(18,2) NULL,
    ValorTotal DECIMAL(18,2) NOT NULL,

    CONSTRAINT FK_CotacaoCobertura_Cotacao FOREIGN KEY (IdCotacao)
        REFERENCES Cotacao(Id)
        ON DELETE CASCADE
);
GO

CREATE TABLE Cobertura (
    Id INT PRIMARY KEY,
    Description NVARCHAR(100) NOT NULL,
    Type NVARCHAR(20) NOT NULL,
    Value DECIMAL(18,2) NOT NULL
);
GO

CREATE TABLE FaixaIdade (
    Id INT PRIMARY KEY,
    Description NVARCHAR(50) NOT NULL,
    Desconto DECIMAL(5,2) NOT NULL,
    Agravo DECIMAL(5,2) NOT NULL
);
GO

-- Tabela: Produto
CREATE TABLE Produto (
    Id INT PRIMARY KEY,
    Description NVARCHAR(100) NOT NULL,
    BaseValue DECIMAL(18,2) NOT NULL,
    Limite DECIMAL(18,2) NOT NULL
);
GO

-- Tabela: Parceiro
CREATE TABLE Parceiro (
    Id INT PRIMARY KEY,
    Description NVARCHAR(100) NOT NULL,
    Secret NVARCHAR(50) NOT NULL
);
GO

-- Tabela: TipoParentesco
CREATE TABLE TipoParentesco (
    Id INT PRIMARY KEY,
    Description NVARCHAR(50) NOT NULL
);
GO
